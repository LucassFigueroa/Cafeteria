COMO REEJECUTAR (desde la carpeta Cafeteria del repositorio)
  Automatizadas:   cd front ; npm start      ->  cd pruebas-selenium ; mvn test -Dheadless=true
  Unitarias:       cd front ; npx ng test --watch=false --coverage
  SonarQube:       powershell -ExecutionPolicy Bypass -File .\ejecutar-sonarqube.ps1
  Sitio estatico:  cd front ; npx ng build --stats-json ; npx serve@14 -s dist/front/browser -l 8080
  OWASP ZAP:       powershell -ExecutionPolicy Bypass -File .\pruebas-seguridad\ejecutar-zap.ps1
  JMeter:          powershell -ExecutionPolicy Bypass -File .\pruebas-carga\ejecutar-jmeter.ps1
